cmake_minimum_required(VERSION 3.25)
project(BrowserMonitor VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BM_BUILD_HOST "Build Windows streaming host" ON)
add_executable(core-tests host/tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE host/include)
add_test(NAME core-tests COMMAND core-tests)
if(BM_BUILD_HOST)
  if(NOT WIN32)
    message(FATAL_ERROR "The host requires Windows 11. Use -DBM_BUILD_HOST=OFF for portable tests.")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
  set(NO_TESTS ON CACHE BOOL "" FORCE)
  set(LIBSRTP_TEST_APPS OFF CACHE BOOL "" FORCE)
  set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  add_subdirectory(.deps/libdatachannel EXCLUDE_FROM_ALL)
  add_executable(browser-monitor host/src/main.cpp host/src/capture.cpp host/src/converter.cpp host/src/encoder.cpp host/src/transport.cpp host/src/wgc.cpp host/src/pattern.cpp)
  target_include_directories(browser-monitor PRIVATE host/include .deps/libdatachannel/deps/json/include)
  target_compile_definitions(browser-monitor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(browser-monitor PRIVATE /W4 /EHsc /permissive- /utf-8)
  target_link_libraries(browser-monitor PRIVATE datachannel-static d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 bcrypt user32 gdi32 shcore windowsapp evr)
  file(READ "${CMAKE_SOURCE_DIR}/shared/ice.json" BM_ICE_JSON)
  configure_file(host/include/ice_config.hpp.in generated/ice_config.hpp @ONLY)
  target_include_directories(browser-monitor PRIVATE "${CMAKE_BINARY_DIR}/generated")
  install(TARGETS browser-monitor RUNTIME DESTINATION bin)
endif()
